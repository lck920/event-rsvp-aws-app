## Table

CREATE TABLE events (
  event_id      VARCHAR(64) PRIMARY KEY,
  title         VARCHAR(200) NOT NULL,
  description   TEXT,
  start_at      DATETIME NOT NULL,
  venue         VARCHAR(200),
  banner_url    VARCHAR(500),
  created_at    TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
